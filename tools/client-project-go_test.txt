func authoredRequest(url string) Request {
    body:="authored POST\r\n雪"
    return Request{Method:"POST",URL:url,Headers:HeaderPairs{},Body:&body,TimeoutMS:500}
}
func TestNetworkErrorAccounting(t *testing.T) {
    var seen atomic.Int32
    server:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){seen.Add(1);w.WriteHeader(200)}))
    defer server.Close()
    input:=authoredRequest(server.URL)
    provenance:=Provenance{Format:"reb-client-project-v1",Mode:"raw"}
    ctx,cancel:=context.WithCancel(context.Background());cancel()
    result,err:=Send(ctx,input,Config{AllowedOrigin:server.URL},provenance,true)
    if err==nil || result.NetworkAttempted || result.NetworkRequests==nil || *result.NetworkRequests!=0 || seen.Load()!=0 {
        t.Fatalf("pre-canceled context must remain network-free: %+v %v",result,err)
    }
    ctx,cancel=context.WithCancel(context.Background());defer cancel()
    release:=make(chan struct{})
    delayed:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
        io.Copy(io.Discard,r.Body);seen.Add(1);cancel();<-release
    }))
    defer delayed.Close()
    input=authoredRequest(delayed.URL)
    result,err=Send(ctx,input,Config{AllowedOrigin:delayed.URL},provenance,true)
    close(release)
    if strings.Contains(result.Instruction,"--send") || !strings.Contains(result.Instruction,"check its state") {t.Fatal("unsafe post-attempt retry instruction")}
    if err==nil || !result.NetworkAttempted || result.NetworkRequests!=nil || seen.Load()!=1 {
        t.Fatalf("canceled observed POST must report attempted and unknown request count: %+v %v",result,err)
    }
    short:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){w.Header().Set("Content-Length","20");io.WriteString(w,"short")}))
    defer short.Close()
    input=authoredRequest(short.URL)
    result,err=Send(context.Background(),input,Config{AllowedOrigin:short.URL},provenance,true)
    if err==nil || !result.NetworkAttempted || result.NetworkRequests==nil || *result.NetworkRequests!=1 || result.Status!=200 || result.URL!=short.URL || result.Body!="short" || !result.BodyTruncated {
        t.Fatalf("failed response read still confirms a sent request: %+v %v",result,err)
    }
}
func TestTLSVerificationStaysEnabled(t *testing.T) {
    var seen atomic.Int32
    server:=httptest.NewUnstartedServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){seen.Add(1)}))
    server.Config.ErrorLog=log.New(io.Discard,"",0)
    server.StartTLS();defer server.Close()
    result,err:=Send(context.Background(),authoredRequest(server.URL),Config{AllowedOrigin:server.URL},Provenance{Format:"reb-client-project-v1",Mode:"raw"},true)
    if err==nil || !result.NetworkAttempted || result.NetworkRequests!=nil || seen.Load()!=0 {
        t.Fatalf("untrusted authored TLS certificate must be refused before HTTP: %+v %v",result,err)
    }
}
func TestStrictSharedJSON(t *testing.T) {
    request:=`{"method":"GET","url":"http://127.0.0.1/","headers":[],"body":"","timeout_ms":100}`
    config:=`{"allowedOrigin":"http://127.0.0.1","request":`+request+`}`
    directory:=t.TempDir();path:=filepath.Join(directory,"input.json")
    for name,data:=range map[string]string{
        "folded method":`{"method":"GET","METHOD":"DELETE","url":"http://127.0.0.1/","headers":[],"body":"","timeout_ms":100}`,
        "duplicate method":`{"method":"GET","method":"DELETE","url":"http://127.0.0.1/","headers":[],"body":"","timeout_ms":100}`,
        "null header":`{"method":"GET","url":"http://127.0.0.1/","headers":[["X",null]],"body":"","timeout_ms":100}`,
        "incomplete":`{"method":"GET"}`,
    } {
        if err:=os.WriteFile(path,[]byte(data),0600);err!=nil {t.Fatal(err)}
        var parsed Request
        if err:=ReadJSON(path,&parsed);err==nil {t.Fatal("accepted "+name)}
    }
    for name,data:=range map[string]string{
        "merge replacement":`{"allowedOrigin":"http://127.0.0.1","request":`+request+`,"request":{"method":"DELETE"}}`,
        "folded origin":`{"allowedOrigin":"http://127.0.0.1","ALLOWedorigin":"http://other.invalid","request":`+request+`}`,
    } {
        if err:=os.WriteFile(path,[]byte(data),0600);err!=nil {t.Fatal(err)}
        var parsed Config
        if err:=ReadJSON(path,&parsed);err==nil {t.Fatal("accepted "+name)}
    }
    if err:=os.WriteFile(path,[]byte(config),0600);err!=nil {t.Fatal(err)}
    var parsed Config
    if err:=ReadJSON(path,&parsed);err!=nil || parsed.Request.Method!="GET" {t.Fatalf("valid shared config rejected: %v",err)}
    if err:=os.WriteFile(path,[]byte(strings.Repeat("[",66)+"0"+strings.Repeat("]",66)),0600);err!=nil {t.Fatal(err)}
    var nested any
    if err:=ReadJSON(path,&nested);err==nil {t.Fatal("accepted excessive JSON nesting")}
    var provenance Provenance
    if err:=json.Unmarshal([]byte(`{"format":"reb-client-project-v1","mode":"redacted_copy","Mode":"raw","problems":[],"future":{"ok":true}}`),&provenance);err!=nil || provenance.Mode!="redacted_copy" {
        t.Fatalf("provenance aliases must not override exact fields: %+v %v",provenance,err)
    }
}

func TestConfigurationAndResponseChecks(t *testing.T) {
    stringValue:=func(value string)*string{return &value}
    number:=func(value int)*int{return &value}
    input:=authoredRequest("http://127.0.0.1")
    valid:=Config{ConfigVersion:"reb-client-config-v1",AllowedOrigin:"http://127.0.0.1",TimeoutMS:number(500),RedirectPolicy:stringValue("manual"),CookiePolicy:stringValue("omit"),
        ResponseChecks:&ResponseChecks{Statuses:[]int{200},ContentType:stringValue("application/json"),JSONFields:[]JSONField{
            {Pointer:"/availability",Type:"string",Equals:json.RawMessage(`"in_stock"`)},
            {Pointer:"/quantity",Type:"number",Equals:json.RawMessage(`4`)},
            {Pointer:"/store/a~1b/~0key",Type:"boolean",Equals:json.RawMessage(`true`)},
            {Pointer:"/variants/0",Type:"null",Equals:json.RawMessage(`null`)},
        }}}
    policy,err:=configure(valid,input);if err!=nil {t.Fatal(err)}
    body:=`{"availability":"in_stock","quantity":4,"store":{"a/b":{"~key":true}},"variants":[null]}`
    response:=func(text string)Result{return Result{Status:200,Headers:[][]string{{"Content-Type","application/json; charset=utf-8"}},Body:text}}
    result:=response(body)
    if err:=ValidateResponse(&result,policy);err!=nil || !result.Validation.Passed {t.Fatalf("valid authored stock response: %+v %v",result,err)}
    cases:=[]struct{name,body,code string;field JSONField}{
        {"missing",body,"response_field_missing",JSONField{Pointer:"/absent",Type:"number"}},
        {"null",`{"quantity":null}`,"response_field_type",JSONField{Pointer:"/quantity",Type:"number"}},
        {"wrong type",body,"response_field_type",JSONField{Pointer:"/quantity",Type:"string"}},
        {"wrong value",body,"response_field_value",JSONField{Pointer:"/quantity",Type:"number",Equals:json.RawMessage(`3`)}},
        {"rounding",`{"quantity":9007199254740991.4}`,"response_field_value",JSONField{Pointer:"/quantity",Type:"number",Equals:json.RawMessage(`9007199254740991`)}},
        {"underflow",`{"quantity":1e-400}`,"response_field_value",JSONField{Pointer:"/quantity",Type:"number",Equals:json.RawMessage(`0`)}},
        {"exponent",`{"quantity":4e0}`,"response_field_value",JSONField{Pointer:"/quantity",Type:"number",Equals:json.RawMessage(`4`)}},
        {"malformed",`{"quantity":`,"response_json",JSONField{Pointer:"/quantity",Type:"number"}},
        {"duplicate",`{"quantity":4,"quantity":0}`,"response_json",JSONField{Pointer:"/quantity",Type:"number"}},
        {"surrogate",`{"quantity":"\ud800"}`,"response_json",JSONField{Pointer:"/quantity",Type:"string"}},
        {"array member",body,"response_field_missing",JSONField{Pointer:"/variants/length",Type:"number"}},
    }
    for _,item:=range cases {t.Run(item.name,func(t *testing.T){
        itemPolicy:=policy;itemPolicy.ResponseChecks=ResponseChecks{JSONFields:[]JSONField{item.field}}
        result:=response(item.body);err:=ValidateResponse(&result,itemPolicy)
        typed,ok:=err.(*ClientError);if !ok || typed.Code!=item.code {t.Fatalf("expected %s, got %v",item.code,err)}
    })}
    for name,mutate:=range map[string]func(*Result){
        "response_status":func(result *Result){result.Status=401},
        "response_content_type":func(result *Result){result.Headers=[][]string{{"Content-Type","text/html"}}},
        "response_truncated":func(result *Result){result.BodyTruncated=true},
        "response_json":func(result *Result){result.BodyInvalidUTF8=true},
    }{result:=response(body);mutate(&result);err:=ValidateResponse(&result,policy);typed,ok:=err.(*ClientError);if !ok || typed.Code!=name {t.Fatalf("expected %s, got %v",name,err)}}
    for _,config:=range []Config{
        {ConfigVersion:"future"},{TimeoutMS:number(99)},{TimeoutMS:number(30001)},{RedirectPolicy:stringValue("follow")},{CookiePolicy:stringValue("browser")},
        {ResponseChecks:&ResponseChecks{Statuses:[]int{200,200}}},{ResponseChecks:&ResponseChecks{ContentType:stringValue("application/json; charset=utf-8")}},
        {ResponseChecks:&ResponseChecks{JSONFields:[]JSONField{{Pointer:"/bad~2",Type:"string"}}}},
        {ResponseChecks:&ResponseChecks{JSONFields:[]JSONField{{Pointer:"/x",Type:"number",Equals:json.RawMessage(`1.5`)}}}},
        {ResponseChecks:&ResponseChecks{JSONFields:[]JSONField{{Pointer:"/x",Type:"object",Equals:json.RawMessage(`{}`)}}}},
    }{if _,err:=configure(config,input);err==nil {t.Fatalf("accepted invalid config: %+v",config)}}
    saved:=browserProfileSend;browserProfileSend=nil;defer func(){browserProfileSend=saved}()
    for _,config:=range []Config{
        {Transport:&TransportConfig{Mode:"browser"}},
        {Transport:&TransportConfig{Mode:"standard",BrowserProfile:&BrowserProfile{Browser:"chrome"}}},
        {Transport:&TransportConfig{Mode:"browser-profile",BrowserProfile:&BrowserProfile{Browser:"chrome",Version:"133.0.0.0",Platform:"linux"}}},
        {Transport:&TransportConfig{Mode:"browser-profile",BrowserProfile:&BrowserProfile{Browser:"chrome",Version:"137.0.0.0",Platform:"linux"}}},
    }{if _,err:=configure(config,input);err==nil {t.Fatalf("accepted unsupported transport: %+v",config)}}
}
