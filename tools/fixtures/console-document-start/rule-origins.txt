# Document-start rule origins: one verdict and one origin per line.
#
# The browser's dependency-free validator (common/native_console_document_start.h,
# run by tools/check-native-console-document-start.cpp) and the backend's
# (apps/origin-trace-backend/src/native_console.rs) both run every line, so they
# accept and refuse exactly the same set. An accepted origin is spelled exactly as
# url::Origin::Serialize writes it; any other spelling would arm a rule that no
# document can ever match, so it is refused instead.

accept http://127.0.0.1:8080
accept https://example.test
accept http://localhost

accept https://255.255.255.255
# A port is part of the origin unless it is the scheme's own default.
accept http://example.test:443
accept https://example.test:80
# IPv6 only as its canonical bracketed literal.
